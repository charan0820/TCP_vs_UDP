cmake_minimum_required(VERSION 3.16)
project(transport_lab LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
execute_process(COMMAND ${PKG_CONFIG_EXECUTABLE} --list-all OUTPUT_VARIABLE available)
string(REGEX MATCH "(lib)?ns3[a-zA-Z0-9_.-]*-core[ \t]" core_match "${available}")
if(NOT core_match)
  message(FATAL_ERROR "NS-3 development libraries not found. Install libns3-dev.")
endif()
string(REGEX REPLACE "-core[ \t]$" "" ns_prefix "${core_match}")
execute_process(COMMAND ${PKG_CONFIG_EXECUTABLE} --modversion ${ns_prefix}-core OUTPUT_VARIABLE ns_version OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT DEFINED TRANSPORT_CXX_STANDARD)
  if(ns_version MATCHES "^3\\.[0-9]+" AND ns_version VERSION_LESS "3.36")
    set(TRANSPORT_CXX_STANDARD 17)
  else()
    set(TRANSPORT_CXX_STANDARD 20)
  endif()
endif()
set(CMAKE_CXX_STANDARD ${TRANSPORT_CXX_STANDARD})
pkg_check_modules(NS3 REQUIRED IMPORTED_TARGET ${ns_prefix}-core ${ns_prefix}-network ${ns_prefix}-internet ${ns_prefix}-point-to-point ${ns_prefix}-applications ${ns_prefix}-traffic-control)
add_executable(transport-lab tcp_udp.cc)
target_link_libraries(transport-lab PRIVATE PkgConfig::NS3)
pkg_check_modules(NETANIM QUIET IMPORTED_TARGET ${ns_prefix}-netanim)
option(TRANSPORT_ENABLE_NETANIM "Link optional NS-3 NetAnim XML exporter" OFF)
if(NOT TRANSPORT_ENABLE_NETANIM)
  set(NETANIM_FOUND FALSE)
endif()
# Ubuntu's 3.35 NetAnim package pulls in a Wi-Fi DSO with an initialization-order
# crash before main(). Keep packet playback available without that optional DSO.
if(ns_version VERSION_LESS "3.36")
  set(NETANIM_FOUND FALSE)
endif()
foreach(include_dir IN LISTS NETANIM_INCLUDE_DIRS)
  if(NOT EXISTS "${include_dir}")
    set(NETANIM_FOUND FALSE)
  endif()
endforeach()
if(NETANIM_FOUND)
  target_link_libraries(transport-lab PRIVATE PkgConfig::NETANIM)
  target_compile_definitions(transport-lab PRIVATE HAVE_NETANIM=1)
  file(WRITE "${CMAKE_BINARY_DIR}/capabilities.json" "{\"netanim\":true}")
else()
  file(WRITE "${CMAKE_BINARY_DIR}/capabilities.json" "{\"netanim\":false}")
endif()
